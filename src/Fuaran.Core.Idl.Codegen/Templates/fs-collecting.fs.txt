// ---------------------------------------------------------------------------
// Phase 377 — COLLECTING DECODERS. Beside every short-circuiting `dec*` decoder above, a `col*`
// decoder answers EVERY defect it finds, as a `DecodeError list`, in one deterministic order:
//   - an object's members in FIELD DECLARATION ORDER (a node: `id`, its kind, then its envelope
//     fields), each member's defects at that member's position, its nested defects included
//     (depth-first);
//   - a list's items in index order, and a map's entries in document order;
//   - a leaf (a scalar, an enum, a sentinel, a hosted slot, verbatim JSON) reports at most one
//     defect, and so does a value of the wrong kind or an absent or unknown discriminator, whose
//     members are never read.
// The first defect of a collecting decoder is the defect its short-circuiting twin reports, and on
// a clean input both answer the same value. A case refine runs once its members all decoded, and a
// host projection supplies only a short-circuiting decoder, so each adds at most one defect of its
// own. Phase 541 — every node, non-generic union, record and kind spec is read through one record
// of decoders, `CollectingDecoders`, so a host's decoder at a type applies at every depth. Phase
// 562 — so is every enumeration, JSON scalar and generic union, and a declared member the object
// does not carry asks the record's `Absent` first. Codes and paths are Core's `DecodeError`; a
// consumer whose specification orders or names defects differently maps them at its own seam.
// ---------------------------------------------------------------------------
let private cOne (r: Result<'T, DecodeError>) : Result<'T, DecodeError list> =
    match r with
    | Ok v -> Ok v
    | Error e -> Error [ e ]

let private cLift (dec: JVal -> Result<'T, DecodeError>) (j: JVal) : Result<'T, DecodeError list> = cOne (dec j)

let private cErrs (r: Result<'T, DecodeError list>) : DecodeError list =
    match r with
    | Ok _ -> []
    | Error es -> es

// One step further from the root, for every defect a member or an item answered.
let private cUnder (step: PathSegment) (r: Result<'T, DecodeError list>) : Result<'T, DecodeError list> =
    match r with
    | Ok v -> Ok v
    | Error es -> Error(es |> List.map (DecodeError.under step))

// The first defect — how a short-circuiting decoder is handed a collecting codec.
let private cFirst (r: Result<'T, DecodeError list>) : Result<'T, DecodeError> =
    match r with
    | Ok v -> Ok v
    | Error(e :: _) -> Error e
    | Error [] -> dFail DecodeCode.SchemaFault "a refusal" "a collecting decoder answered no defect and no value"

let private cObj (j: JVal) : Result<(string * JVal) list, DecodeError list> = cOne (dObj j)

let private cList (dec: JVal -> Result<'T, DecodeError list>) (j: JVal) : Result<'T list, DecodeError list> =
    match j with
    | JArr xs ->
        let results = xs |> List.mapi (fun i x -> dec x |> cUnder (PathSegment.Index i))

        match results |> List.collect cErrs with
        | [] -> Ok(results |> List.choose (function Ok v -> Some v | Error _ -> None))
        | errors -> Error errors
    | other -> cOne (Error(Decoder.wrongKind "array" other))

// Every entry is checked, in document order; a repeated key keeps its FIRST value, as `dMap` does.
let private cMap (dec: JVal -> Result<'T, DecodeError list>) (j: JVal) : Result<Map<string, 'T>, DecodeError list> =
    match j with
    | JObj fs ->
        let results = fs |> List.map (fun (k, v) -> k, (dec v |> cUnder (PathSegment.Key k)))

        match results |> List.collect (snd >> cErrs) with
        | [] ->
            (Map.empty, results)
            ||> List.fold (fun items (k, r) ->
                match r with
                | Ok d when not (Map.containsKey k items) -> Map.add k d items
                | _ -> items)
            |> Ok
        | errors -> Error errors
    | other -> cOne (Error(Decoder.wrongKind "object" other))

// The node's own framing (its `kind` member) read under its key: absent is `dReq`'s refusal.
let private cReq (name: string) (fs: (string * JVal) list) (dec: JVal -> Result<'T, DecodeError list>) : Result<'T, DecodeError list> =
    match fs |> List.tryFind (fun (k, _) -> k = name) with
    | Some(_, v) -> dec v |> cUnder (PathSegment.Key name)
    | None -> cOne (dReq name fs (fun _ -> dFail DecodeCode.SchemaFault "never read" "never read"))

// Phase 562 — a DECLARED member the object `j` does not carry: the record's `Absent` is asked first,
// with the holder's name and the member's key. `Ok None` is the reading the member's presence rule
// gives absence (`none`); `Ok(Some v)` reads `v` as the member; a refusal is the member's own, at its
// key. The default `Absent` answers `Ok None`, so a record that overrides nothing reads as before.
let private cAbsent
    (ov: CollectingDecoders)
    (holder: string)
    (j: JVal)
    (name: string)
    (dec: JVal -> Result<'T, DecodeError list>)
    (none: unit -> Result<'T, DecodeError list>)
    : Result<'T, DecodeError list> =
    match ov.Absent ov holder name j with
    | Ok None -> none ()
    | Ok(Some v) -> dec v |> cUnder (PathSegment.Key name)
    | Error es -> Error(es |> List.map (DecodeError.under (PathSegment.Key name)))

// A required declared member: absent (and the hook answering `None`) is `dReq`'s refusal.
let private cReqAt (ov: CollectingDecoders) (holder: string) (j: JVal) (name: string) (fs: (string * JVal) list) (dec: JVal -> Result<'T, DecodeError list>) : Result<'T, DecodeError list> =
    match fs |> List.tryFind (fun (k, _) -> k = name) with
    | Some(_, v) -> dec v |> cUnder (PathSegment.Key name)
    | None -> cAbsent ov holder j name dec (fun () -> cOne (dReq name fs (fun _ -> dFail DecodeCode.SchemaFault "never read" "never read")))

let private cOptAt (ov: CollectingDecoders) (holder: string) (j: JVal) (name: string) (fs: (string * JVal) list) (dec: JVal -> Result<'T, DecodeError list>) : Result<'T option, DecodeError list> =
    match fs |> List.tryFind (fun (k, _) -> k = name) with
    | Some(_, v) -> dec v |> cUnder (PathSegment.Key name) |> Result.map Some
    | None -> cAbsent ov holder j name (dec >> Result.map Some) (fun () -> Ok None)

let private cDefAt (ov: CollectingDecoders) (holder: string) (j: JVal) (name: string) (fs: (string * JVal) list) (dec: JVal -> Result<'T, DecodeError list>) (dflt: 'T) : Result<'T, DecodeError list> =
    match fs |> List.tryFind (fun (k, _) -> k = name) with
    | Some(_, v) -> dec v |> cUnder (PathSegment.Key name)
    | None -> cAbsent ov holder j name dec (fun () -> Ok dflt)

// Phase 562 — the generated decoders at the JSON scalars, each the short-circuiting reader lifted.
let private colString (_: CollectingDecoders) (j: JVal) : Result<string, DecodeError list> = cOne (dStr j)

let private colInt (_: CollectingDecoders) (j: JVal) : Result<int, DecodeError list> = cOne (dInt j)

let private colFloat (_: CollectingDecoders) (j: JVal) : Result<float, DecodeError list> = cOne (dFloat j)

let private colBool (_: CollectingDecoders) (j: JVal) : Result<bool, DecodeError list> = cOne (dBool j)

// Phase 562 — the generated `Absent`: every absent member reads by its presence rule.
let private colAbsent (_: CollectingDecoders) (_: string) (_: string) (_: JVal) : Result<JVal option, DecodeError list> = Ok None
